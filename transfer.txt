#ifndef MAINVIEW_H
#define MAINVIEW_H

#include <QGraphicsView>
#include <QWheelEvent>
#include <QMouseEvent>
#include <QMenu>
#include <QAction>
#include <QDesktopServices>
#include <QUrl>
#include <QCursor>

#include "model/SceneCarte.hpp"

/**
 * @class MainView
 * @brief Vue personnalisée pour l'affichage et l'interaction avec la carte.
 * * Cette classe gère le rendu de la scène de la carte, le zoom, le déplacement,
 * ainsi que les interactions spécifiques comme le menu contextuel sur les villes.
 */
 class MainView : public QGraphicsView {
	Q_OBJECT
	public :
		/**
		 * @brief Constructeur de MainView.
		 * @param scene Pointeur vers la scène à afficher.
		 * @param w Parent de la vue.
		 * @param orientation_nord Angle de rotation de l'indicateur du Nord.
		 */
		MainView(SceneCarte *scene, QWidget *w, int orientation_nord = 0)
            : QGraphicsView(scene, w), orientation_nord(orientation_nord) 
        {
			Q_UNUSED(scene);
			// Inverse l'axe Y pour correspondre aux coordonnées géographiques standards
			scale(1, -1); 
			setMouseTracking(true); 
            // Force la mise à jour complète pour éviter les artefacts sur les éléments fixes (rose des vents)
			setViewportUpdateMode(QGraphicsView::FullViewportUpdate); 
		}

		~MainView(){}

	signals:
		/** @brief Signal émis lors du changement de zone visible. */
		void coord_viewport(QRectF);
		/** @brief Signal émis lors du mouvement de la souris (coordonnées scène). */
		void position(QPointF);

	protected:
		/** @brief Ajuste la vue lors du redimensionnement de la fenêtre. */
		void resizeEvent (QResizeEvent*) override {
			if (this->transform().m11() == 1) {
				this->fitInView(sceneRect(), Qt::KeepAspectRatio);
			}
		}

		/** @brief Gère le zoom avec la molette de la souris. */
		void wheelEvent(QWheelEvent *event) override{
			int angle = event->angleDelta().y();
			qreal facteur_zoom = (angle > 0) ? 1.1 : 0.9;
			scale(facteur_zoom, facteur_zoom);
		}

		/** @brief Intercepte le rendu pour calculer et émettre les limites de la vue actuelle. */
		void paintEvent (QPaintEvent * event) override {
			// Calcul du rectangle visible converti en coordonnées scène
			QPolygonF poly = mapToScene(viewport()->rect());
			emit coord_viewport(poly.boundingRect());
            
			QGraphicsView::paintEvent(event);
		}

		/** @brief Suit le mouvement de la souris et émet sa position. */
		void mouseMoveEvent(QMouseEvent *event) override{
			emit position(mapToScene(event->pos()));
			QGraphicsView::mouseMoveEvent(event);
		}

		/** @brief Gère le clic droit pour ouvrir le menu contextuel Wikipédia. */
		void mousePressEvent(QMouseEvent *event) override{
			if(event->button() == Qt::RightButton){
				QPointF pos_scene = mapToScene(event->pos());
				QGraphicsItem *item = scene()->itemAt(pos_scene, transform());

				if(item) {
					// On suppose que la donnée 0 contient le nom de la ville
					QString villeName = item->data(0).toString();
					if(!villeName.isEmpty()) {
						QMenu contextMenu;
						QAction *openWikipediaAction = new QAction("Afficher le wikipédia de la ville", this);
						contextMenu.addAction(openWikipediaAction);

						connect(openWikipediaAction, &QAction::triggered, this, [villeName](){
							QString url = QString("https://fr.wikipedia.org/wiki/%1").arg(villeName);
							QDesktopServices::openUrl(QUrl(url));
						});
						
						contextMenu.exec(QCursor::pos());
					}
				} else {
					QGraphicsView::mousePressEvent(event);
				}
			}
		}

		/** @brief Dessine l'arrière-plan (Océan) et l'indicateur fixe du Nord. */
		void drawBackground(QPainter *painter, const QRectF &) override {
			// 1. Dessin de l'eau en coordonnées viewport (pixel fixe)
			painter->setWorldMatrixEnabled(false);
			painter->setBrush(QBrush(QColor(70, 130, 180))); 
			painter->setPen(Qt::NoPen);
			painter->drawRect(viewport()->rect());
			
			// 2. Dessin de l'icône Nord
			QPixmap pixmap("../ressource/images/north.png");
			if (!pixmap.isNull()) {
                QTransform matrice;
                matrice.rotate(orientation_nord);
                QPixmap pixmapRotated = pixmap.transformed(matrice);
                painter->drawPixmap(10, 10, pixmapRotated); // Position fixe en haut à gauche
            }
            
			painter->setWorldMatrixEnabled(true);
		}

		/** @brief Dessine les éléments d'interface utilisateur par-dessus la scène (Barre d'échelle). */
		void drawForeground(QPainter *painter, const QRectF &) override {
			painter->setTransform(QTransform(), false); // Passage en coordonnées écran
			painter->setPen(QPen(Qt::black, 2));
			painter->setFont(QFont("Arial", 10, QFont::Bold));
		
			// Configuration graphique de l'échelle (bas droite)
			int scaleBarLength = 50;
			int margin = 15;
			int x2 = this->width() - margin;
			int y2 = this->height() - margin;
			int x1 = x2 - scaleBarLength;
			
			// Dessin de la ligne et des taquets
			painter->drawLine(x1, y2, x2, y2);
			painter->drawLine(x1, y2 - 3, x1, y2 + 3);
			painter->drawLine(x2, y2 - 3, x2, y2 + 3);
		
			painter->drawText(QRect(x1, y2 - 25, scaleBarLength, 20), Qt::AlignCenter, "10 km");
		}

	private:
		int orientation_nord; 
};

#endif
